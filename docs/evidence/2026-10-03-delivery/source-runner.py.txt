def note(kind, **fields):
    row = dict(time=time.time(),kind=kind,**fields)
    with (DATA/'actions.jsonl').open('a') as f: f.write(json.dumps(row)+'\n')
    print('VERIFY '+json.dumps(row), flush=True)
def run_command(cmd):
    op=cmd['op']
    note('command', **cmd)
    if op=='connect':
        ctrl.connect()
        note('connected',connected=ctrl.driver is not None and ctrl.box is not None,version=ctrl.vv_version)
    elif op=='replay':
        handler(cmd['text'],True,{})
        note('replay_complete',text=cmd['text'],last_sent=ctrl.last_sent)
    elif op=='pair':
        handler('John chapter three verse sixteen',True,{})
        note('pair_first_complete',last_sent=ctrl.last_sent)
        time.sleep(cmd.get('delay',5))
        handler('John chapter three verse seventeen',True,{})
        note('pair_second_complete',last_sent=ctrl.last_sent)
    elif op=='config':
        if 'confidence' in cmd: engine.CONFIDENCE_THRESHOLD=cmd['confidence']
        if 'manual' in cmd: engine.REQUIRE_MANUAL_CONFIRM=cmd['manual']
        note('config',confidence=engine.CONFIDENCE_THRESHOLD,manual=engine.REQUIRE_MANUAL_CONFIRM)
    elif op=='disconnect':
        if ctrl.driver: ctrl.driver.quit()
        ctrl.driver=ctrl.box=ctrl.btn=None
        engine.REMOTE_URL='http://127.0.0.1:59999/control.html'
    elif op=='restore_target': engine.REMOTE_URL=safe['remote_url']
    elif op=='snapshot':
        (DATA/'events.json').write_text(json.dumps(engine.get_delivery_events(),indent=2))
        note('snapshot',events=len(engine.get_delivery_events()),last_sent=ctrl.last_sent,
             queue=len(engine.verse_queue),diag_state=app._delivery_diag_box.cget('state'))
    elif op=='quit':
        if ctrl.driver: ctrl.driver.quit()
        app.after(0,app.destroy)
    else: note('unknown_command')
def reader():
    for line in sys.stdin:
        try: commands.put(json.loads(line))
        except Exception as ex: note('input_error',error=type(ex).__name__)
def pump():
    while not commands.empty():
        cmd=commands.get()
        threading.Thread(target=run_command,args=(cmd,),daemon=True).start()
    app.after(100,pump)
threading.Thread(target=reader,daemon=True).start()
note('source',engine=engine.__file__,gui=gui.__file__,engine_sha256=hashlib.sha256((ROOT/'vv_streaming_master.py').read_bytes()).hexdigest(),gui_sha256=hashlib.sha256((ROOT/'vv_gui.py').read_bytes()).hexdigest())
app.after(100,pump)
app.mainloop()
